   12 -- Template substitution engine built on Parsec. Supports parameterized
   13 -- open\/close delimiters for embedding named placeholders in strings and files.
   14 -- The default syntax uses @{{>>=name}}@ but custom delimiters (e.g. @{{::=name}}@,
   15 -- @\{%name%\}@) are supported via the @*With@ variants.

   48 -- | Like 'Text.Parsec.manyTill' but returns both the accumulated results and
   49 -- the end value as a pair.
   50 --
   51 -- @since 0.1.0.0
   52 manyTill_ :: ParsecT s u m a -> ParsecT s u m end -> ParsecT s u m ([a], end)
   53 manyTill_ p end = go
   54   where
   55     go = (([],) <$> end) <|> liftA2 (\x (xs, y) -> (x : xs, y)) p go
   56

   58 -- | Parse zero or more occurrences of @inside@ between @open@ and @close@,
   59 -- returning the collected inner results.
   60 --
   61 -- @since 0.1.0.0
   62 between' :: Stream s m Char => ParsecT s u m open -> ParsecT s u m end -> ParsecT s u m a -> ParsecT s u m [a]
   63 between' open close inside = do
   64   _ <- open
   65   (x, _) <- manyTill_ inside close
   66   pure x
   67 
   68 
   69 -- | Parse exactly one occurrence of @match@ between @open@ and @end@.
   70 --
   71 -- @since 0.1.0.0
   72 between1 :: Stream s m Char => ParsecT s u m open -> ParsecT s u m end -> ParsecT s u m a -> ParsecT s u m a
   73 between1 open end match = open *> match <* end

   76 -- | Parser for valid template names: starts with letter, underscore, or hyphen;
   77 -- followed by alphanumeric, underscore, or hyphen.
   78 --
   79 -- @since 0.1.0.0
   80 validName :: Stream s m Char => ParsecT s u m Name
   81 validName = do
   82   first_ <- letter <|> char '_' <|> char '-'
   83   rest <- many $ alphaNum <|> char '_' <|> char '-'
   84   pure $ first_ : rest
   85

   87 -- | A template placeholder name (e.g. @\"name\"@ in @{{>>=name}}@).
   88 --
   89 -- @since 0.1.0.0
   90 type Name = String
   91 
   92 -- | Template parser parameterized by open\/close delimiters.
   93 --
   94 -- @templateParserWith \"{{>>=\" \"}}\"@ is equivalent to 'templateParser'.
   95 --
   96 -- @since 0.1.0.0
   97 templateParserWith :: Stream s m Char => String -> String -> ParsecT s u m Name
   98 templateParserWith open close =
   99   between1 (string open) (string close) validName
  100 
  101 -- | Default template parser using @{{>>=name}}@ syntax.
  102 --
  103 -- @since 0.1.0.0
  104 templateParser :: Stream s m Char => ParsecT s u m Name
  105 templateParser = templateParserWith "{{>>=" "}}"

  117 -- | Replace all template occurrences in the input using the map,
  118 -- parameterized by open\/close delimiters.
  119 --
  120 -- Calls 'error' if a placeholder references a name not present in the map.
  121 --
  122 -- @since 0.1.0.0
  123 unTemplateWith :: String -> String -> Map.Map String String -> String -> String
  124 unTemplateWith open close tmplMap input = streamEdit (templateParserWith open close) mapLookup input
  125   where
  126     mapLookup :: Name -> String
  127     mapLookup n = case Map.lookup n tmplMap of
  128       Just v -> v
  129       Nothing -> error $! "name " <> open <> n <> close <> " not defined"
  130 
  131 -- | Replace all @{{>>=name}}@ occurrences in the input using the map.
  132 --
  133 -- Calls 'error' if a placeholder references a name not present in the map.
  134 --
  135 -- @since 0.1.0.0
  136 unTemplate :: Map.Map String String -> String -> String
  137 unTemplate = unTemplateWith "{{>>=" "}}"
  138 
  139 -- | Like 'unTemplateWith' but reads from a file. Parameterized by delimiters.
  140 --
  141 -- TODO: make this catchable as an IOError
  142 --
  143 -- @since 0.1.0.0
  144 unTemplateFileWith :: String -> String -> Map.Map String String -> FilePath -> IO String
  145 unTemplateFileWith open close tmplMap fp = do
  146   s <- readFile fp
  147   pure $ unTemplateWith open close tmplMap s
  148 
  149 -- | Like 'unTemplate' but reads from a file.
  150 --
  151 -- TODO: make this catchable as an IOError
  152 --
  153 -- @since 0.1.0.0
  154 unTemplateFile :: Map.Map String String -> FilePath -> IO String
  155 unTemplateFile = unTemplateFileWith "{{>>=" "}}"
