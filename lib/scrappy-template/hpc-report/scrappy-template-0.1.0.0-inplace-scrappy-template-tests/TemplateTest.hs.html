   35 -- | @since 0.1.0.0
   36 parseOnly :: String -> Either String Name
   37 parseOnly input =
   38   let !s = "" :: String
   39   in  case parse (templateParser <* eof) s input of
   40         Left err -> let !msg = show err in Left msg
   41         Right n  -> Right n
   42 
   43 -- | @since 0.1.0.0
   44 parseOnlyWith :: String -> String -> String -> Either String Name
   45 parseOnlyWith open close input =
   46   let !s = "" :: String
   47   in  case parse (templateParserWith open close <* eof) s input of
   48         Left err -> let !msg = show err in Left msg
   49         Right n  -> Right n
   50 
   51 -- | @since 0.1.0.0
   52 parseValidName :: String -> Either String Name
   53 parseValidName input =
   54   let !s = "" :: String
   55   in  case parse (validName <* eof) s input of
   56         Left err -> let !msg = show err in Left msg
   57         Right n  -> Right n

   59 -- | Check whether an Either is Left, handling both branches.
   60 --
   61 -- @since 0.1.0.0
   62 isLeft :: Either a b -> Bool
   63 isLeft (Left _) = True
   64 isLeft (Right _) = False
   65 
   66 -- | Force a string to its full spine and return it.
   67 --
   68 -- @since 0.1.0.0
   69 forceStr :: String -> String
   70 forceStr [] = []
   71 forceStr (x:xs) = x `seq` forceStr xs `seq` (x:xs)
   72 {-# INLINE forceStr #-}
