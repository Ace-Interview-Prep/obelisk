   37 -- | Generate a valid template name: starts with letter\/underscore\/hyphen,
   38 -- followed by alphanumeric\/underscore\/hyphen.
   39 --
   40 -- @since 0.1.0.0
   41 genValidName :: Gen Name
   42 genValidName = do
   43   first_ <- oneof
   44     [ elements (['a'..'z'] ++ ['A'..'Z'])
   45     , pure '_'
   46     , pure '-'
   47     ]
   48   n <- choose (0, 20 :: Int)
   49   rest <- vectorOf n $ oneof
   50     [ elements (['a'..'z'] ++ ['A'..'Z'] ++ ['0'..'9'])
   51     , pure '_'
   52     , pure '-'
   53     ]
   54   pure (first_ : rest)
   55 
   56 -- | Generate a name that starts with a digit (always invalid).
   57 --
   58 -- @since 0.1.0.0
   59 genDigitStartName :: Gen String
   60 genDigitStartName = do
   61   d <- elements ['0'..'9']
   62   n <- choose (0, 10 :: Int)
   63   rest <- vectorOf n $ elements (['a'..'z'] ++ ['0'..'9'] ++ ['_', '-'])
   64   pure (d : rest)
   65 
   66 -- | Generate a pair of safe delimiters that contain no name-chars
   67 -- (letters, digits, underscore, hyphen) to avoid ambiguity with validName.
   68 --
   69 -- @since 0.1.0.0
   70 genSafeDelimiterPair :: Gen (String, String)
   71 genSafeDelimiterPair = elements
   72   [ ("{{>>=", "}}")
   73   , ("{{::=", "}}")
   74   , ("{%", "%}")
   75   , ("<<", ">>")
   76   , ("[", "]")
   77   , ("<<<START>>>", "<<<END>>>")
   78   , ("((=", "))")
   79   , ("{|", "|}")
   80   ]
   81 
   82 -- | Generate a non-colliding delimiter pair — one that does NOT start with @{{>>=@
   83 -- so it won't accidentally match the default syntax.
   84 --
   85 -- @since 0.1.0.0
   86 genNonDefaultDelimiterPair :: Gen (String, String)
   87 genNonDefaultDelimiterPair = elements
   88   [ ("{{::=", "}}")
   89   , ("{%", "%}")
   90   , ("<<", ">>")
   91   , ("[", "]")
   92   , ("((=", "))")
   93   , ("{|", "|}")
   94   ]
   95 
   96 -- | Generate a plain text string guaranteed not to contain the default
   97 -- template opening delimiter @{{>>=@.
   98 --
   99 -- @since 0.1.0.0
  100 genPlainText :: Gen String
  101 genPlainText = do
  102   n <- choose (0, 50 :: Int)
  103   vectorOf n (elements safeChars)
  104     `suchThat` (\s -> not ("{{>>=" `isInfixOf` s))
  105   where
  106     safeChars :: [Char]
  107     safeChars = ['a'..'z'] ++ ['A'..'Z'] ++ ['0'..'9'] ++ " .,;:!?+-*/"
  108 
  109 -- | Generate a plain text string guaranteed not to contain a given
  110 -- delimiter pair as substrings.
  111 --
  112 -- @since 0.1.0.0
  113 genPlainTextFor :: String -> String -> Gen String
  114 genPlainTextFor open close = do
  115   n <- choose (0, 50 :: Int)
  116   vectorOf n (elements safeChars)
  117     `suchThat` (\s -> not (open `isInfixOf` s) && not (close `isInfixOf` s))
  118   where
  119     safeChars :: [Char]
  120     safeChars = ['a'..'z'] ++ ['A'..'Z'] ++ ['0'..'9'] ++ " .,;:!?+-*/"
  121

  127 -- | Force a string to its full spine and return it.
  128 --
  129 -- @since 0.1.0.0
  130 forceString :: String -> String
  131 forceString [] = []
  132 forceString (x:xs) = x `seq` forceString xs `seq` (x:xs)
  133 {-# INLINE forceString #-}

  136 -- =========================================================================
  137 -- Properties
  138 -- =========================================================================
  139 
  140 -- | @since 0.1.0.0
  141 spec :: Spec
  142 spec = do
  143 
  144   -- =======================================================================
  145   -- validName
  146   -- =======================================================================
  147   describe "Property: validName" $ do
  148 
  149     prop "roundtrip: generated valid names parse back to themselves" $
  150       forAll genValidName $ \name ->
  151         let !s = "" :: String
  152         in  parse (validName <* eof) s name == Right name
  153 
  154     prop "rejects digit-start names" $
  155       forAll genDigitStartName $ \name ->
  156         let !name' = forceString name
  157             !s = "" :: String
  158             !p = validName <* eof
  159             result = parse p s name'
  160         in  case result of
  161               Left _  -> True
  162               Right _ -> False
  163 
  164     prop "rejects empty string" $
  165       property $
  166         let !s = "" :: String
  167             !p = validName <* eof
  168             result = parse p s ""
  169         in  case result of
  170               Left _  -> True
  171               Right _ -> False
  172 
  173     prop "first char of generated name is letter, underscore, or hyphen" $
  174       forAll genValidName $ \name ->
  175         let !name' = forceString name
  176             validFirst = ['a'..'z'] ++ ['A'..'Z'] ++ ['_', '-']
  177         in  case name' of
  178               (c:_) -> c `elem` validFirst
  179               []    -> error "genValidName produced empty string"
  180 
  181     prop "rest chars of generated name are alphanumeric, underscore, or hyphen" $
  182       forAll genValidName $ \name ->
  183         let !name' = forceString name
  184             validRest = ['a'..'z'] ++ ['A'..'Z'] ++ ['0'..'9'] ++ ['_', '-']
  185         in  case name' of
  186               (_:rest) -> all (`elem` validRest) rest
  187               []       -> error "genValidName produced empty string"
  188 
  189   -- =======================================================================
  190   -- templateParser
  191   -- =======================================================================
  192   describe "Property: templateParser" $ do
  193 
  194     prop "roundtrip: wrapping a valid name in {{>>= ... }} parses to the name" $
  195       forAll genValidName $ \name ->
  196         let !s = "" :: String
  197         in  parse (templateParser <* eof) s ("{{>>=" ++ name ++ "}}") == Right name
  198 
  199     prop "roundtrip with arbitrary safe delimiters" $
  200       forAll genSafeDelimiterPair $ \(open, close) ->
  201       forAll genValidName $ \name ->
  202         let !s = "" :: String
  203             !close' = forceString close
  204         in  parse (templateParserWith open close' <* eof) s (open ++ name ++ close') == Right name
  205 
  206     prop "mismatched delimiters cause parse failure" $
  207       forAll genValidName $ \name ->
  208         let !name' = forceString name
  209             !s = "" :: String
  210             !close' = forceString "}}"
  211             !input = forceString ("{{>>=" ++ name' ++ "}}")
  212             result = parse (templateParserWith "{{::=" close' <* eof) s input
  213         in  case result of
  214               Left _  -> True
  215               Right _ -> False
  216 
  217     prop "empty name between delimiters fails" $
  218       forAll genSafeDelimiterPair $ \(open, close) ->
  219         let !close' = forceString close
  220             !s = "" :: String
  221             !input = forceString (open ++ close')
  222             result = parse (templateParserWith open close' <* eof) s input
  223         in  case result of
  224               Left _  -> True
  225               Right _ -> False
  226 
  227     prop "digit-start name between delimiters fails" $
  228       forAll genSafeDelimiterPair $ \(open, close) ->
  229       forAll genDigitStartName $ \name ->
  230         let !name' = forceString name
  231             !close' = forceString close
  232             !s = "" :: String
  233             !input = forceString (open ++ name' ++ close')
  234             result = parse (templateParserWith open close' <* eof) s input
  235         in  case result of
  236               Left _  -> True
  237               Right _ -> False
  238 
  239   -- =======================================================================
  240   -- unTemplate
  241   -- =======================================================================
  242   describe "Property: unTemplate" $ do
  243 
  244     prop "single key substitution" $
  245       forAll genValidName $ \k ->
  246       forAll genPlainText $ \v ->
  247         unTemplate (Map.singleton k v) ("{{>>=" ++ k ++ "}}") == v
  248 
  249     prop "multiple keys: concatenated placeholders produce concatenated values" $
  250       forAll genValidName $ \k1 ->
  251       forAll genValidName $ \k2 ->
  252       k1 /= k2 ==>
  253       forAll genPlainText $ \v1 ->
  254       forAll genPlainText $ \v2 ->
  255         let m = Map.fromList [(k1, v1), (k2, v2)]
  256             input = "{{>>=" ++ k1 ++ "}}{{>>=" ++ k2 ++ "}}"
  257         in  unTemplate m input == v1 ++ v2
  258 
  259     prop "no re-expansion: substitution value containing delimiters is literal" $
  260       forAll genValidName $ \k1 ->
  261       forAll genValidName $ \k2 ->
  262       k1 /= k2 ==>
  263         let dangerousValue = "{{>>=" ++ k2 ++ "}}"
  264         in  unTemplate (Map.singleton k1 dangerousValue) ("{{>>=" ++ k1 ++ "}}") == dangerousValue
  265 
  266     prop "plain text without placeholders passes through unchanged" $
  267       forAll genPlainText $ \s ->
  268         let !m = (Map.empty :: Map.Map String String)
  269         in  Map.size m `seq` unTemplate m s == s
  270 
  271     prop "empty input always yields empty output" $
  272       forAll genValidName $ \k ->
  273       forAll genPlainText $ \v ->
  274         let !k' = forceString k
  275             !v' = forceString v
  276             !m = Map.fromList [(k', v')]
  277         in  Map.size m `seq` unTemplate m "" == ""
  278 
  279     prop "surrounding text is preserved around placeholder" $
  280       forAll genValidName $ \k ->
  281       forAll genPlainText $ \v ->
  282       forAll genPlainText $ \prefix ->
  283       forAll genPlainText $ \suffix ->
  284         unTemplate (Map.singleton k v) (prefix ++ "{{>>=" ++ k ++ "}}" ++ suffix)
  285           == prefix ++ v ++ suffix
  286 
  287     prop "repeated same placeholder yields repeated value" $
  288       forAll genValidName $ \k ->
  289       forAll genPlainText $ \v ->
  290       forAll (choose (1, 10 :: Int)) $ \n ->
  291         let m = Map.singleton k v
  292             input = concat (replicate n ("{{>>=" ++ k ++ "}}"))
  293         in  unTemplate m input == concat (replicate n v)
  294 
  295   -- =======================================================================
  296   -- unTemplateWith (delimiter isolation)
  297   -- =======================================================================
  298   describe "Property: unTemplateWith" $ do
  299 
  300     prop "single key with custom delimiters" $
  301       forAll genSafeDelimiterPair $ \(open, close) ->
  302       forAll genValidName $ \k ->
  303       forAll (genPlainTextFor open close) $ \v ->
  304         unTemplateWith open close (Map.singleton k v) (open ++ k ++ close) == v
  305 
  306     prop "default placeholders ignored by non-default delimiters" $
  307       forAll genNonDefaultDelimiterPair $ \(open, close) ->
  308       forAll genValidName $ \k ->
  309         let !close' = forceString close
  310             !m = (Map.empty :: Map.Map String String)
  311             input = "{{>>=" ++ k ++ "}}"
  312         in  Map.size m `seq` unTemplateWith open close' m input == input
  313 
  314     prop "custom placeholders ignored by default unTemplate" $
  315       forAll genNonDefaultDelimiterPair $ \(open, close) ->
  316       forAll genValidName $ \k ->
  317         let input = open ++ k ++ close
  318             !m = (Map.empty :: Map.Map String String)
  319         in  not ("{{>>=" `isInfixOf` input) ==>
  320             Map.size m `seq` unTemplate m input == input
  321 
  322     prop "surrounding text preserved with custom delimiters" $
  323       forAll genSafeDelimiterPair $ \(open, close) ->
  324       forAll genValidName $ \k ->
  325       forAll (genPlainTextFor open close) $ \v ->
  326       forAll (genPlainTextFor open close) $ \prefix ->
  327       forAll (genPlainTextFor open close) $ \suffix ->
  328         unTemplateWith open close (Map.singleton k v)
  329           (prefix ++ open ++ k ++ close ++ suffix)
  330           == prefix ++ v ++ suffix
  331 
  332   -- =======================================================================
  333   -- manyTill_
  334   -- =======================================================================
  335   describe "Property: manyTill_" $ do
  336 
  337     prop "n copies of 'a' followed by 'b' yields (replicate n 'a', 'b')" $
  338       forAll (choose (0, 100 :: Int)) $ \n ->
  339         let input = replicate n 'a' ++ "b"
  340             !s = "" :: String
  341             result = parse (manyTill_ (char 'a') (char 'b')) s input
  342         in  result == Right (replicate n 'a', 'b')
  343 
  344   -- =======================================================================
  345   -- Cross-cutting
  346   -- =======================================================================
  347   describe "Property: cross-cutting" $ do
  348 
  349     prop "two-pass substitution: default then custom on same string" $
  350       forAll genValidName $ \kDefault ->
  351       forAll genValidName $ \kCustom ->
  352       forAll genPlainText $ \vDefault ->
  353       forAll genPlainText $ \vCustom ->
  354         let input = "{{>>=" ++ kDefault ++ "}} and {{::=" ++ kCustom ++ "}}"
  355             pass1 = unTemplate (Map.singleton kDefault vDefault) input
  356             pass2 = unTemplateWith "{{::=" "}}" (Map.singleton kCustom vCustom) pass1
  357             !msg1 = forceString ("pass1 = " ++ show pass1)
  358             !msg2 = forceString ("pass2 = " ++ show pass2)
  359         in  counterexample msg1 $
  360             counterexample msg2 $
  361             pass2 == vDefault ++ " and " ++ vCustom
